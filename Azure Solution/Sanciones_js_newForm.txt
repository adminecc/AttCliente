function checkUserGroup() {
    $().SPServices({
        operation: "GetGroupCollectionFromUser",
        userLoginName: $().SPServices.SPGetCurrentUser(),
        async: false,
        completefunc: function (xData, Status) {
            if ($(xData.responseXML).find("Group[Name='Atención al Cliente']").length == 1) {
                isAttCliente = true;
            }
            if (($(xData.responseXML).find("Group[Name='Pem Técnicos de Operaciones']").length == 1) || ($(xData.responseXML).find("Group[Name='Pem Operadores de Línea']").length == 1) ||
                ($(xData.responseXML).find("Group[Name='Pem Coordinadores']").length == 1)) {
                isOperaciones = true;
            }
            if (($(xData.responseXML).find("Group[Name='Pem Técnicos de Operaciones']").length == 1) || ($(xData.responseXML).find("Group[Name='Pem Operadores de Línea']").length == 1)) {
                isTecnicoOperador = true;
            }

        }
    });
}

function setValidateFields() {
    // Validacion numerica para los campos numericos
    $("input[Title='Código de la infracción']").keypress(function(event) {
        return event.charCode >= 48 && event.charCode <= 57;
    });
    $("input[Title='Nº Fiscalizador']").keypress(function(event) {
        return event.charCode >= 48 && event.charCode <= 57;
    });
    $("input[Title='Nº SAP']").keypress(function(event) {
        return event.charCode >= 48 && event.charCode <= 57;
    });
    $("input[Title='Importe']").keypress(function(event) {
        return ((event.charCode >= 48 && event.charCode <= 57) || (event.charCode == 44));
    });
    $("input[Title='Nº de Intervenciones']").keypress(function(event) {
        return event.charCode >= 48 && event.charCode <= 57;
    });
    $("input[Title='Nº Reclamación']").keypress(function (event) {
        return event.charCode >= 48 && event.charCode <= 57;
    });


    //hh:mm
    $("input[Title='Hora(estación)']").inputmask("hh:mm", {
        placeholder: "hh:mm",
        insertMode: false,
        showMaskOnHover: false
    });

    $("input[Title='Hora de Inicio']").inputmask("hh:mm", {
        placeholder: "hh:mm",
        insertMode: false,
        showMaskOnHover: false
    });

    $("input[Title='Hora de Fin']").inputmask("hh:mm", {
        placeholder: "hh:mm",
        insertMode: false,
        showMaskOnHover: false
    });


    // Deshabilitamos inicialmente el campo Motivos Anulacion, hasta que se active la casilla de anulacion infraccion
    $("textarea[Title='Motivo Anulación']").prop('disabled', true);
    $("input[Title='Anulación infracción']").change(function () {
        if($(this)[0].checked)
            $("textarea[Title='Motivo Anulación']").prop('disabled', false);
        else
            $("textarea[Title='Motivo Anulación']").prop('disabled', true);
    });
};

function showModalDialogTipoSolicitud() {
    var title = "Nueva Solicitud";

        '<div class="row"><div class="companeros col col-md-12">' +
        '<div class="col col-md-5"><span>Por favor seleccione el tipo de solicitud</span></div>' +

        '<option value="Intervención">Intervención</option>' +
        '<option value="Sanción">Sanción</option></select></div>';


    html += '</div></div></div></div>';

        title: title, html: html, width: '650px', confirmButtonColor: "#008033", allowOutsideClick: false

function removeSectionFromField(startColumn) {
    /*Parametros:
        sectionShortName: Short name para IDs de elementos no puede contener espacios o caracteres especiales.
        sectionLongName: Display name de la cabecera.
        startColumn: Primera columna de la sección.
        endColumn: Última columna de la sección.
      Los dos nombres de columnas deben ser el Display name, NO  el internal name p.ej "Nombre Producto" y no InternalName, p.ej., "Nombre_x0020_Producto"
    */

    // Busca la prim fila de la sección y borra la cabecera
    var sectionStartRow = findFormField(startColumn).closest("tr");
    $(sectionStartRow).prev(".demo-section-header").remove();
};

function setupSection(sectionShortName, sectionLongName, startColumn, endColumn) {
    /*Parametros:
        sectionShortName: Short name para IDs de elementos no puede contener espacios o caracteres especiales.
        sectionLongName: Display name de la cabecera.
        startColumn: Primera columna de la sección.
        endColumn: Última columna de la sección.
      Los dos nombres de columnas deben ser el Display name, NO  el internal name p.ej "Nombre Producto" y no InternalName, p.ej., "Nombre_x0020_Producto"
    */

    // Busca la prim y ult fila de la sección y añade la cabecera
    var sectionStartRow = findFormField(startColumn).closest("tr");
    var sectionEndRow = findFormField(endColumn).closest("tr");

    var thisSection = null;

    if (sectionStartRow[0].innerHTML == sectionEndRow[0].innerHTML) {
        $(sectionStartRow)
          .before("<tr class='demo-section-header' id='" + sectionShortName + "'>" +
            "<td class='demo-section-header' colspan=2>" + sectionLongName + "</td>" +
            "</tr>");
        thisSection = sectionStartRow;
        $(sectionStartRow).attr("section-name", sectionShortName);
        $(sectionEndRow).find("td").addClass("demo-section-last-row-cell");
    } else {
        $(sectionStartRow)
          .before("<tr class='demo-section-header' id='" + sectionShortName + "'>" +
            "<td class='demo-section-header' colspan=2>" + sectionLongName + "</td>" +
            "</tr>");
        thisSection = $(sectionStartRow).nextUntil(sectionEndRow);
        $(sectionStartRow).attr("section-name", sectionShortName);
        $(sectionStartRow).nextUntil(sectionEndRow).attr("section-name", sectionShortName);
        $(sectionEndRow).attr("section-name", sectionShortName);
        $(sectionEndRow).find("td").addClass("demo-section-last-row-cell");
    }

    // Oculta todas las filas de la sección
    $("tr[section-name='" + sectionShortName + "']").css("display", "none");

    // Añadimos el comportamiento del click en la cabecera
    $("tr#" + sectionShortName).click(function () {
        var thisSectionRows = $("tr[section-name='" + sectionShortName + "']");
        if ($(this).next("tr[section-name='" + sectionShortName + "']").css("display") == "block") {
            $(thisSectionRows).css("display", "none");
            $(this).removeClass("demo-collapse");
        } else {
            $(thisSectionRows).css("display", "block");
            $(this).addClass("demo-collapse");
            if (isOperaciones) {
                $('td.ms-formlabel:contains("Nombre del Cliente")').parent().hide();
                $('td.ms-formlabel:contains("DNI")').parent().hide();
                $('td.ms-formlabel:contains("Nombre del Tutor")').parent().hide();
                $('td.ms-formlabel:contains("DNI del Tutor")').parent().hide();
                $('td.ms-formlabel:contains("Domicilio")').parent().hide();
                $('td.ms-formlabel:contains("Teléfono")').parent().hide();
                $('td.ms-formlabel:contains("Comunicado a AOP")').parent().hide();
            }
        }
    });

    // Expande la sección si hay algun error de validación
    $("tr[section-name='" + sectionShortName + "'] .ms-formbody .ms-formvalidation").each(function () {
        if ($(this).html().length > 0) {
            $("tr[section-name='" + sectionShortName + "']").css("display", "block");
            $("tr#" + sectionShortName).addClass("demo-collapse");
            // No need to look at any more rows
            return false;
        }
    });

    // Expand the section if it's been passed on the Query String
    if (sectionShortName == expandSection) {
        $("tr[section-name='" + sectionShortName + "']").css("display", "block");
        $("tr#" + sectionShortName).addClass("demo-collapse");
    }
};

// This function (borrowed from SPServices) finds a column's formBody in the page
function findFormField(columnName) {
    var thisFormBody;
    // There's no easy way to find one of these columns; we'll look for the comment with the columnName
    var searchText = RegExp("FieldName=\"" + columnName.replace(/[-[\]{}()*+?.,\\^$|#\s]/g, "\\$&") + "\"", "gi");
    // Loop through all of the ms-formbody table cells
    $("td.ms-formbody").each(function () {
        // Check for the right comment
        if (searchText.test($(this).html())) {
            thisFormBody = $(this);
            // Found it, so we're done
            return false;
        }
    });
    return thisFormBody;
};

function addFieldToSection(columnName, endColumnName,sectionName) {
    // Busca la ultima fila de la seccion
    $($("input[title='" + columnName + "']").parent().parent().parent().insertAfter($("input[title='" + endColumnName + "']").parent().parent().parent()));

    // Modificamos los atributos
    $("input[title='" + columnName + "']").parent().parent().parent().css('display','none');
    $("input[title='" + columnName + "']").parent().parent().parent().attr('section-name',sectionName);


};
